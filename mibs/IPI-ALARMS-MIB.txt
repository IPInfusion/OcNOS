ipiAlarmsAlarmTable OBJECT-TYPE 
	SYNTAX SEQUENCE OF IpiAlarmsAlarmEntry
	MAX-ACCESS not-accessible
	STATUS current
	DESCRIPTION 
		"This table provides information for List of alarms, keyed by a unique id"
	::= { ipi-alarms 1 }

ipiAlarmsAlarmEntry OBJECT-TYPE
	SYNTAX IpiAlarmsAlarmEntry
	MAX-ACCESS not-accessible
	STATUS current
	DESCRIPTION 
		"An entry containing information for List of alarms, keyed by a unique id"
	INDEX { ipiAlarmsAlarmId }
	::= { ipiAlarmsAlarmTable 1 }

IpiAlarmsAlarmEntry::= SEQUENCE { 
	ipiAlarmsAlarmId OCTET STRING,
	ipiAlarmsAlarmResource OCTET STRING,
	ipiAlarmsAlarmText OCTET STRING,
	ipiAlarmsAlarmTimeCreated TimeTicks,
	ipiAlarmsAlarmAlarmReportedTimestamp OCTET STRING,
	ipiAlarmsAlarmAlarmSeverity INTEGER,
	ipiAlarmsAlarmTypeId INTEGER,
	ipiAlarmsAlarmIsClear INTEGER	}

ipiAlarmsAlarmId OBJECT-TYPE
	SYNTAX OCTET STRING (SIZE (0..1965))
	MAX-ACCESS read-only
	STATUS current
	DESCRIPTION
		"References the unique alarm id"
	::= { ipiAlarmsAlarmEntry 1 }

ipiAlarmsAlarmResource OBJECT-TYPE
	SYNTAX OCTET STRING (SIZE (0..1965))
	MAX-ACCESS read-only
	STATUS current
	DESCRIPTION
		"The item that is under alarm within the device"
	::= { ipiAlarmsAlarmEntry 2 }

ipiAlarmsAlarmText OBJECT-TYPE
	SYNTAX OCTET STRING (SIZE (0..1965))
	MAX-ACCESS read-only
	STATUS current
	DESCRIPTION
		"The string used to inform operators about the alarm"
	::= { ipiAlarmsAlarmEntry 3 }

ipiAlarmsAlarmTimeCreated OBJECT-TYPE
	SYNTAX TimeTicks
	UNITS "milliseconds"
	MAX-ACCESS read-only
	STATUS current
	DESCRIPTION
		"The time at which the alarm was raised by the system. This value is expressed relative to the Unix Epoch"
	::= { ipiAlarmsAlarmEntry 4 }

ipiAlarmsAlarmAlarmReportedTimestamp OBJECT-TYPE
	SYNTAX OCTET STRING (SIZE (0..1965))
	MAX-ACCESS read-only
	STATUS current
	DESCRIPTION
		"The date-time at which the alarm was raised by the system"
	::= { ipiAlarmsAlarmEntry 5 }

ipiAlarmsAlarmAlarmSeverity OBJECT-TYPE
	SYNTAX INTEGER {critical(0), major(1), minor(2), warning(3), unknown(4)}
	MAX-ACCESS read-only
	STATUS current
	DESCRIPTION
		"The severity level indicating the criticality and impact of the alarm"
	::= { ipiAlarmsAlarmEntry 6 }

ipiAlarmsAlarmTypeId OBJECT-TYPE
	SYNTAX INTEGER {ais(0), eqpt(1), los(2), ots(3), opwr(4), unknown(5)}
	MAX-ACCESS read-only
	STATUS current
	DESCRIPTION
		"The abbreviated name of the alarm"
	::= { ipiAlarmsAlarmEntry 7 }

ipiAlarmsAlarmIsClear OBJECT-TYPE
	SYNTAX INTEGER { false(0), true(1) }
	MAX-ACCESS read-only
	STATUS current
	DESCRIPTION
		"Indicates if this alarm is a clear alarm or not"
	::= { ipiAlarmsAlarmEntry 8 }
